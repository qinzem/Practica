#include <iostream>   подключаем библиотеку для ввода вывода

int main() {  главная функция
    int a = 4, b = 5; переменные целочисленные

    std::cout << "a = " << a << ", b = " << b << "\n"; вывод начальных значений а и б
    a += b; получаем 9
    b = a - b; получаем 5
    a -= b; получаем 4

    return 0; заканчиваем выполнение функции
}

int main() { главная функция
    std::vector<int> A = {1, 3, 5, 7, 9}; создаем вектор(массив А) с 5 элементами
    std::vector<int> B = {2, 4, 6, 8}; создаем вектор (массив Б) с 4 элементами

    size_t i = 0, j = 0;(начальные курсоры по массивам а и б) i-для А j- для B
    size_t n = A.size(), m = B.size(); (возвращаем количество элементов в каждом массиве

    while (i < n || j < m) {(выполняем цикл пока выполняется условие что текущий курсор меньше каждого из массивов) или A или B
        if (j == m || (i < n && A[i] <= B[j])) (когда элементы в массивах есть)
            std::cout << A[i++] << " "; (читаем текцщий элемент массива и увеличиваем на 1)
        else (когда в А элементы кончились или текущий элемент в Б меньше)
            std::cout << B[j++] << " ";
    }

    return 0;(завершаем выполнение)
}

#include <iostream> подключаем библиотеку для ввода вывода
#include <vector> подключаем библиотеку вектор (умные массивы которые знают свой размер)

int main() {главная функция
    int n = 5; переменная для правильной суммы
    std::vector<int> arr = {0, 1, 2, 4, 5}; массив

    long long sum = 0; сюда накапливаем сумму всех чисел из массива
    for (int x : arr) для каждого числа из массива arr)
        sum += x; заполняем переменную суммы(суммой всех элементов)

    long long expected = n * n / 2;(формула Гауса) для вычисления пропущенного числа

    std::cout << "Пропущено: " << expected - sum << "\n"; ожидаемая сумма всех чисел = фактическая сумма + пропущенное число.

    return 0;завершаем выполнение
}
